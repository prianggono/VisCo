cmake_minimum_required(VERSION 3.24)
project(VisCoNativeHost LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(visco-native-host main.cpp capture-runtime.cpp runtime-adapters.cpp network-frame-runtime.cpp output-runtime.cpp asio-audio-runtime.cpp audio-mixer-runtime.cpp)
target_compile_definitions(visco-native-host PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(visco-native-host PRIVATE advapi32 d3d11 d3dcompiler dxgi dxguid mf mfplat mfuuid mfreadwrite ole32 uuid avrt ws2_32 mmdevapi windowscodecs)
if(WIN32)
  target_link_options(visco-native-host PRIVATE /SUBSYSTEM:WINDOWS)
endif()
if(MSVC)
  target_compile_options(visco-native-host PRIVATE /W4 /permissive-)
endif()
