cmake_minimum_required(VERSION 3.24)
project(VisCoNativeHost LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(visco-native-host main.cpp)
target_compile_definitions(visco-native-host PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(visco-native-host PRIVATE d3d11 dxgi dxguid mf mfplat mfuuid mfreadwrite ole32 uuid avrt ws2_32 mmdevapi)
if(MSVC)
  target_compile_options(visco-native-host PRIVATE /W4 /permissive-)
endif()
